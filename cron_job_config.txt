# we made a cron job to run on our wpi linux server
# it runs the check_and_recovery.sh script every minute
# this checks that the chatbot is still running 

# if it detects an outage, it will initiate a series of escalating recovery steps
# at the highest level of recovery, it will run deploy_first_part and deploy_second_part
# this re-secures the virtual machine and redeploys the app

# the recovery log from check_and_recovery is available at ~/.local/state/group16-recovery/recovery.log

#but the cron job also appends the whatever is printed by the recovery file to  ~/.local/state/group16-recovery/cron.log

# this is the cron job configuration details:

SHELL=/bin/bash
PATH=/usr/local/bin:/usr/bin:/bin

 * * * * * /home/my_wpi_username/case_study_2/check_and_recovery.sh >> /home/my_wpi_username/.local/state/group16-recovery/cron.log 2>&1
